import { initializeApp } from "firebase/app"; import { getDatabase, ref, set } from "firebase/database"; // 1. Firebase 설정 (알려주신 URL 적용) const firebaseConfig = { databaseURL: "https://challenge-90fcd-default-rtdb.asia-southeast1.firebasedatabase.app/", // projectId, apiKey 등 다른 설정값들이 있다면 함께 넣어주세요. }; // 2. 앱 및 데이터베이스 초기화 const app = initializeApp(firebaseConfig); const db = getDatabase(app); // 3. 데이터 저장 함수 생성 function writeUserData(userId, name, email) { // 'users/유저ID' 경로에 데이터 저장 set(ref(db, 'users/' + userId), { username: name, email: email }) .then(() => { console.log("데이터 저장 성공!"); }) .catch((error) => { console.error("데이터 저장 실패: ", error); }); } // 실행 테스트 writeUserData("user123", "홍길동", "hong@example.com");
